  def compute_delta_action(
        self,
        target_ee_pose: np.ndarray,
        # --- Arguments that must be passed from the live simulation ---
        model: mujoco.MjModel,
        data: mujoco.MjData,
        ee_site_id: int,
        joint_qpos_indices: np.ndarray,
        effective_dt: float,
        max_dq: float,
    ) -> np.ndarray:
        """
        [DEFINITIVE STABLE PD CONTROLLER VERSION]
        Computes a normalized delta action using a robust, damped Proportional-Derivative
        (PD) controller in task space, translated to joint velocities via dIK.
        """
        damping = 1e-4 # Damping for the dIK solver

        # --- 1. GET CURRENT STATE FROM LIVE SIMULATION DATA ---
        current_ee_pos = data.site_xpos[ee_site_id]
        current_ee_mat = data.site_xmat[ee_site_id].reshape(3, 3)
        current_ee_quat = R.from_matrix(current_ee_mat).as_quat()
        
        # Calculate current end-effector velocity (twist)
        jac_pos = np.zeros((3, model.nv))
        jac_rot = np.zeros((3, model.nv))
        ee_body_id = model.site_bodyid[ee_site_id]
        mujoco.mj_jac(model, data, jac_pos, jac_rot, current_ee_pos, ee_body_id)
        
        J_full = np.vstack([jac_pos, jac_rot])
        J = J_full[:, joint_qpos_indices]
        current_qvel = data.qvel[joint_qpos_indices]
        current_ee_vel = J @ current_qvel

        # --- 2. IMPLEMENT STABLE PD CONTROL LAW ---
        # Gains tuned for responsiveness and stability
        Kp = 40.0  # Proportional gain (FAST response to error)
        Kd = 1.3   # Derivative gain (DAMPING to prevent overshoot/oscillation)

        pos_error = target_ee_pose[:3] - current_ee_pos
        orn_error = R.from_quat(target_ee_pose[3:]) * R.from_quat(current_ee_quat).inv()
        orn_error_vec = orn_error.as_rotvec()

        # Proportional term
        p_term = Kp * np.concatenate([pos_error, orn_error_vec])
        
        # Derivative term
        d_term = -Kd * current_ee_vel

        ee_vel_target = p_term + d_term

        # --- 3. SOLVE FOR JOINT VELOCITIES (dIK) ---
        try:
            J_T_J = J.T @ J
            regularized_term = damping * np.eye(J.shape[1])
            lhs = J_T_J + regularized_term
            rhs = J.T @ ee_vel_target
            dq = np.linalg.solve(lhs, rhs)
        except np.linalg.LinAlgError:
            return np.zeros(len(joint_qpos_indices))

        # --- 4. NORMALIZE AND RETURN ACTION ---
        action = np.clip(dq / max_dq, -1.0, 1.0)
        return action
