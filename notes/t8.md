# FILE: scripts/train_hybrid.py --> inside run_hybrid_training function

    # ... after the preparation section ...

    print("\n--- Starting Hybrid Training Loop ---")

    # --- 3.1: Define Training Parameters ---
    # How often we run the PPO learning before doing BC updates
    rl_steps_per_iteration = 4096 # e.g., two PPO rollouts
    # How many BC gradient steps to perform after each RL iteration
    bc_updates_per_iteration = 10 
    # The weight of the BC loss (lambda_bc from the explanation)
    # We will start it at 1.0 and anneal it down to 0.
    initial_bc_lambda = 1.0
    
    total_timesteps = args.total_timesteps
    start_time = time.time()

    # --- 3.2: The Main Loop ---
    for iteration in range(total_timesteps // rl_steps_per_iteration):
        current_timesteps = ppo_agent.num_timesteps
        
        # --- PHASE A: Reinforcement Learning ---
        # Run the standard PPO learn method for a fixed number of steps
        ppo_agent.learn(
            total_timesteps=current_timesteps + rl_steps_per_iteration,
            reset_num_timesteps=False, # IMPORTANT: Do not reset the counter
            log_interval=1 # We will do custom logging
        )
        
        # --- PHASE B: Imitation Learning (BC Regularization) ---
        # Calculate the current BC loss weight (annealing schedule)
        progress = current_timesteps / total_timesteps
        current_bc_lambda = initial_bc_lambda * (1 - progress)

        total_bc_loss = 0.0
        if current_bc_lambda > 0:
            for _ in range(bc_updates_per_iteration):
                try:
                    # Get a batch of expert data
                    obs_expert, act_expert = next(expert_iterator)
                except StopIteration:
                    # Reset iterator if the dataset ends (for non-infinite datasets)
                    expert_iterator = iter(expert_loader)
                    obs_expert, act_expert = next(expert_iterator)
                
                # Move data to the same device as the policy
                act_expert = act_expert.to(ppo_agent.policy.device)
                obs_expert_device = {}
                for key, val in obs_expert.items():
                    obs_expert_device[key] = val.to(ppo_agent.policy.device)
                
                # Get the policy's predicted action for the expert observations
                # We need to call the policy's forward pass directly
                predicted_actions, _, _ = ppo_agent.policy(obs_expert_device)
                
                # Calculate the BC loss (Mean Squared Error)
                bc_loss = bc_loss_fn(predicted_actions, act_expert)
                
                # Apply the annealing weight
                weighted_bc_loss = bc_loss * current_bc_lambda
                
                # Perform the gradient update
                bc_optimizer.zero_grad()
                weighted_bc_loss.backward()
                torch.nn.utils.clip_grad_norm_(policy_params_to_update, 1.0)
                bc_optimizer.step()
                
                total_bc_loss += bc_loss.item()
        
        # --- 3.3: Logging ---
        avg_bc_loss = total_bc_loss / bc_updates_per_iteration if bc_updates_per_iteration > 0 else 0
        
        # Use SB3's logger to record our custom metrics
        ppo_agent.logger.record("custom/bc_loss", avg_bc_loss)
        ppo_agent.logger.record("custom/bc_lambda", current_bc_lambda)
        
        # Dump all logs (RL stats from the last `learn` call + our custom ones)
        ppo_agent.logger.dump(step=current_timesteps)

        print(f"Iter {iteration+1} | Timesteps: {current_timesteps} | "
              f"BC Lambda: {current_bc_lambda:.3f} | Avg BC Loss: {avg_bc_loss:.5f}")

        # Optional: Save intermediate checkpoints
        if (iteration + 1) % 10 == 0:
            save_path = run_dir / "checkpoints" / f"hybrid_policy_{current_timesteps}.zip"
            ppo_agent.save(save_path)

### Summary of Changes and Why It's the "Best Practice"

1.  **What We Did:** We created a high-level training loop that alternates between letting SB3's PPO run its on-policy learning and injecting our own supervised "imitation" updates.
2.  **Why it Works:** The PPO `learn` call is responsible for **exploration** and learning from the sparse task rewards (`R_grasp`, `R_lift`, etc.). The BC update phase acts as a **regularizer** or "safety harness," pulling the policy back towards the expert's behavior, which prevents it from "forgetting" how to reach the cube while it's trying to learn the difficult grasping part.
3.  **Annealing `lambda_bc`:** The weight of the BC loss (`current_bc_lambda`) slowly decreases over time. This is a crucial detail.
    *   **At the beginning:** The BC loss is strong, forcing the agent to stay very close to the expert.
    *   **Towards the end:** The BC loss becomes zero, allowing the agent to fully rely on the RL reward signal to find a policy that might be even *better* than the original expert.

This hybrid approach robustly combines the strengths of both imitation learning and reinforcement learning, giving you the highest probability of successfully training a competent and reliable policy.
